	# The whole loop starts here. Money and attention, one country at a time, until there
	# is enough of a network there to be worth using.
	POR_cultivar_a_rede = {
		icon = GFX_decision_generic_political_discourse

		cost = 25
		days_re_enable = 60

		targets = { AGL UNI MOZ GUB VER SAO TIM CAB EGU }
		targets_dynamic = yes
		target_array = POR_network_target_array

		target_root_trigger = {
			has_country_flag = POR_shadow_doors_open
			NOT = { has_country_flag = POR_stood_down }
		}

		target_trigger = {
			FROM = {
				exists = yes
				is_subject = no
				NOT = { has_war_with = ROOT }
				NOT = { is_subject_of = ROOT }
				check_variable = { POR_net@ROOT < 1 }
			}
			is_in_array = { POR_network_target_array = FROM.id }
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = POR_cultivar_a_rede_tt
				check_variable = { treasury > 1.4 }
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision POR_cultivar_a_rede [FROM.GetName]"
			set_temp_variable = { treasury_change = -1.5 }
			modify_treasury_effect = yes

			FROM = { POR_score_client = yes }

			set_temp_variable = { POR_net_step = 0.10 }
			if = {
				limit = { has_country_flag = POR_os_amigos_do_regime }
				add_to_temp_variable = { POR_net_step = 0.05 }
			}
			multiply_temp_variable = { POR_net_step = POR_reward_mult }

			FROM = {
				add_to_variable = { POR_net@ROOT = ROOT.POR_net_step }
				clamp_variable = { var = POR_net@ROOT min = 0 max = 1 }
				add_to_variable = { POR_heat@ROOT = 1 }
			}

			set_temp_variable = { percent_change = 1.5 }
			multiply_temp_variable = { percent_change = POR_reward_mult }
			set_temp_variable = { tag_index = ROOT.id }
			set_temp_variable = { influence_target = FROM.id }
			change_influence_percentage = yes

			if = {
				limit = { has_country_flag = POR_via_aberta }
				set_temp_variable = { POR_heat_add = 1 }
				POR_add_exposure = yes
			}
		}

		ai_will_do = {
			base = 5
			modifier = {
				factor = 3
				FROM = { influence_higher_20 = yes }
			}
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}
		}
	}

		icon = GFX_decision_generic_operation

			FROM = {

				is_subject = no

				NOT = { is_in_faction_with = ROOT }
				NOT = { is_subject_of = ROOT }
				NOT = { has_government = ROOT }
				NOT = { has_country_flag = POR_client_cooldown@ROOT }
				check_variable = { POR_net@ROOT > 0.699 }
			}
			is_in_array = { POR_network_target_array = FROM.id }
		}

		available = {
			FROM = {
				influence_higher_30 = yes
				has_civil_war = no
			}
			has_equipment = { infantry_weapons_1 > 999 }
			custom_trigger_tooltip = {
				tooltip = POR_o_pronunciamento_local_tt
				check_variable = { treasury > 5.9 }
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision POR_o_pronunciamento_local [FROM.GetName]"
			set_temp_variable = { treasury_change = -6 }
			modify_treasury_effect = yes
			add_equipment_to_stockpile = {
				type = infantry_weapons_1
				amount = -1000
			}

			set_country_flag = POR_pronunciamento_running

			FROM = {
				POR_score_client = yes
				add_to_variable = { POR_heat@ROOT = 15 }
				set_country_flag = { flag = POR_pronunciamento_pending@ROOT days = 60 value = 1 }
				country_event = { id = portugal.20 days = 3 }

		}

				limit = { NOT = { country_exists = FROM } }
				custom_effect_tooltip = POR_pronunciamento_aborted_tt
			}
			else_if = {
				limit = { FROM = { has_civil_war = yes } }
				custom_effect_tooltip = POR_pronunciamento_aborted_tt
			}
			else = { POR_start_client_civil_war = yes }
		}

		ai_will_do = {
			base = 3
			modifier = {
				factor = 0
				check_variable = { POR_shadow_exposure > 54 }
			}
			modifier = {
				factor = 0
				check_variable = { interest_rate > 8 }
			}
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}
