#

# conversion hooks for any fascist civil war spawn (00_on_actions.txt:2354), so party
# slots 10 and 11 are never installed by this file.

	set_variable = { POR_exposure_tier = 0 }
	set_variable = { POR_exposure_breach_bonus = 0 }
	set_variable = { POR_ops_cap = 2 }
	set_variable = { POR_active_operations = 0 }

	set_variable = { POR_ideological_development = 0 }
	set_variable = { POR_ideology_tier = 0 }
	set_variable = { POR_party_flips = 0 }
	set_variable = { POR_fiscal_path = 0 }

	set_variable = { POR_social_path = 0 }
	set_variable = { POR_deficit_path = 0 }
	set_variable = { POR_banking_path = 0 }
	set_variable = { POR_energy_path = 0 }
	set_variable = { POR_obras_path = 0 }
	set_variable = { POR_labour_path = 0 }

	set_variable = { POR_education_path = 0 }
	set_variable = { POR_housing_path = 0 }
	set_variable = { POR_costumes_path = 0 }
	POR_update_blocs = yes
	POR_watch_party_flip = yes

	clear_array = POR_network_target_array
	clear_array = POR_evacuation_array
	if = { limit = { country_exists = AGL } add_to_array = { POR_network_target_array = AGL.id } }

	if = { limit = { country_exists = VER } add_to_array = { POR_network_target_array = VER.id } }
	if = { limit = { country_exists = SAO } add_to_array = { POR_network_target_array = SAO.id } }
	if = { limit = { country_exists = TIM } add_to_array = { POR_network_target_array = TIM.id } }

	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = POR_shadow_network } } }

	}

	# Everybody in the community can see they are in it.
	every_other_country = {

			portuguese_speaking = yes
			NOT = { has_idea = POR_comunidade_lusofona_idea }
		}
		add_ideas = POR_comunidade_lusofona_idea
	}
}

# Scores one country Portugal is working on. Runs in that country's scope with ROOT = POR,
# and writes its two results back as temp variables on ROOT.
#
# Underdog pays, pariah pays more and costs more. Backing the weak and the outlawed is how
# a small country buys a large position, and the bill is exposure, not money.
POR_score_client = {
	set_temp_variable = { ROOT.POR_udg = 0 }
	if = { limit = { has_idea = non_power } add_to_temp_variable = { ROOT.POR_udg = 3 } }
	else_if = { limit = { has_idea = minor_power } add_to_temp_variable = { ROOT.POR_udg = 2 } }
	else_if = { limit = { has_idea = regional_power } add_to_temp_variable = { ROOT.POR_udg = 1 } }

	if = {
		limit = { check_variable = { deployed_army_manpower_k < ROOT.deployed_army_manpower_k } }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}
	if = {
		limit = { check_variable = { ROOT.gdp_total > gdp_total } }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}
	if = {
		limit = { has_war = yes surrender_progress > 0.20 }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}

	set_temp_variable = { ROOT.POR_prh = 0 }
	if = { limit = { has_idea = non_state_actor } add_to_temp_variable = { ROOT.POR_prh = 3 } }
	else_if = { limit = { has_idea = lacks_international_recognition } add_to_temp_variable = { ROOT.POR_prh = 2 } }

	if = { limit = { has_idea = Massive_International_Sanctions } add_to_temp_variable = { ROOT.POR_prh = 3 } }
	else_if = { limit = { has_idea = international_sanctions } add_to_temp_variable = { ROOT.POR_prh = 2 } }
	else_if = {
		limit = { OR = { has_idea = Western_Sanctions has_idea = Reduced_Western_Sanctions } }
		add_to_temp_variable = { ROOT.POR_prh = 1 }
	}

	if = {
		limit = {
			OR = {
				has_idea = unsc_restricted_volunteers_too
				has_dynamic_modifier = { modifier = united_nations_security_council_sanctions }
			}
		}
		add_to_temp_variable = { ROOT.POR_prh = 2 }
	}

	ROOT = {
		clamp_temp_variable = { var = POR_udg min = 0 max = 6 }
		clamp_temp_variable = { var = POR_prh min = 0 max = 6 }

		# Reward multiplier, 1.00 to 2.45.
		set_temp_variable = { POR_reward_mult = POR_udg }
		multiply_temp_variable = { POR_reward_mult = 0.10 }
		set_temp_variable = { POR_prh_term = POR_prh }
		multiply_temp_variable = { POR_prh_term = 0.10 }
		add_to_temp_variable = { POR_reward_mult = POR_prh_term }
		add_to_temp_variable = { POR_reward_mult = 1 }
		if = {
			limit = { has_completed_focus = POR_a_causa_dos_oprimidos }
			add_to_temp_variable = { POR_reward_mult = 0.25 }
		}
		clamp_temp_variable = { var = POR_reward_mult min = 1 max = 2.45 }

		# Exposure multiplier, 1.00 to 2.50. Only the pariah term feeds it: backing a losing
		# side is not a scandal in Brussels, backing an outlawed one is.
		set_temp_variable = { POR_heat_mult = POR_prh }
		multiply_temp_variable = { POR_heat_mult = 0.25 }
		add_to_temp_variable = { POR_heat_mult = 1 }
	}
}

# Raises exposure by POR_heat_add, after scaling it by the target's pariah standing.
# Expects POR_heat_add set, and runs in POR's scope with the target already scored.
POR_add_exposure = {
	# POR_heat_mult only exists after POR_score_client has run. Callers that are not scoring a
	# client - an annexation, a coup in Madrid, a garrison left on a beach - were multiplying
	# by an unset temp and adding nothing.
	if = {
		limit = { check_variable = { POR_heat_mult = 0 } }
		set_temp_variable = { POR_heat_mult = 1 }
	}
	multiply_temp_variable = { POR_heat_add = POR_heat_mult }
	add_to_variable = { POR_shadow_exposure = POR_heat_add }
	clamp_variable = { var = POR_shadow_exposure min = 0 max = 100 }
}

# Walks the exposure meter to its tier and applies what Brussels does about it.
POR_update_exposure_tier = {
	clamp_variable = { var = POR_shadow_exposure min = 0 max = 100 }

	if = { limit = { check_variable = { POR_shadow_exposure > 74 } } set_variable = { POR_exposure_tier = 4 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 54 } } set_variable = { POR_exposure_tier = 3 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 34 } } set_variable = { POR_exposure_tier = 2 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 14 } } set_variable = { POR_exposure_tier = 1 } }
	else = { set_variable = { POR_exposure_tier = 0 } }

	if = {
		limit = { check_variable = { POR_exposure_tier = 0 } }
		set_variable = { POR_exposure_breach_bonus = 0 }
		if = {
			limit = { has_dynamic_modifier = { modifier = POR_eu_exposure_modifier } }
			remove_dynamic_modifier = { modifier = POR_eu_exposure_modifier }
		}
	}
	else = {
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = POR_eu_exposure_modifier } } }
			add_dynamic_modifier = { modifier = POR_eu_exposure_modifier }
		}

		set_variable = { POR_exposure_trade_opinion_factor = POR_exposure_tier }
		multiply_variable = { POR_exposure_trade_opinion_factor = -0.05 }
		set_variable = { POR_exposure_political_power_factor = POR_exposure_trade_opinion_factor }

		set_variable = { POR_exposure_foreign_influence_defense_modifier = 0 }
		if = {
			limit = { check_variable = { POR_exposure_tier > 1 } }
			set_variable = { POR_exposure_foreign_influence_defense_modifier = -0.05 }
		}

		set_variable = { POR_exposure_interest_rate_multiplier_modifier = 0 }
		set_variable = { POR_exposure_stability_factor = 0 }
		set_variable = { POR_exposure_breach_bonus = 0 }

		if = {
			limit = { check_variable = { POR_exposure_tier = 3 } }
			set_variable = { POR_exposure_interest_rate_multiplier_modifier = 1.0 }
			set_variable = { POR_exposure_breach_bonus = 4 }
		}
		else_if = {
			limit = { check_variable = { POR_exposure_tier = 4 } }
			set_variable = { POR_exposure_interest_rate_multiplier_modifier = 2.0 }
			set_variable = { POR_exposure_stability_factor = -0.05 }
			set_variable = { POR_exposure_breach_bonus = 6 }
		}
	}
}

# Weekly hygiene. Every derived counter is recomputed from the arrays rather than
# decremented, so an annexed or recycled client cannot leave a counter stuck.
POR_prune_shadow_arrays = {
	clear_temp_array = POR_dropped_ids

	if = {
		limit = { check_variable = { POR_shadow_clients^num > 0 } }
		for_each_scope_loop = {
			array = POR_shadow_clients
			if = {
				limit = {
					OR = {
						NOT = { exists = yes }
						NOT = { has_country_flag = POR_shadow_client }
					}
				}
				add_to_temp_array = { PREV.POR_dropped_ids = THIS.id }
			}
		}
		for_each_loop = {
			array = POR_dropped_ids
			value = v
			remove_from_array = { POR_shadow_clients = v }
		}
	}

	clear_temp_array = POR_dropped_ids

	if = {
		limit = { check_variable = { POR_lusophone_subjects^num > 0 } }
		for_each_scope_loop = {
			array = POR_lusophone_subjects
			if = {
				limit = {
					OR = {
						NOT = { exists = yes }
						NOT = { is_subject_of = PREV }
					}
				}
				add_to_temp_array = { PREV.POR_dropped_ids = THIS.id }
			}
		}
		for_each_loop = {
			array = POR_dropped_ids
			value = v
			remove_from_array = { POR_lusophone_subjects = v }
		}
	}

	set_variable = { POR_active_operations = POR_shadow_clients^num }
}

# Starts the rebellion. ROOT is Portugal, FROM is the country we have been working on.
#
# Never calls start_civil_war_against_the_strongest_opposition or its sibling: both find
# the rebel with random_enemy_country limited to original_tag = ROOT
# (01_civil_war_effects.txt:118), which never matches when the instigator is a third party.
# Ideology tokens are literal for the same reason; ideology = ROOT inside FROM resolves to
# the instigator. No capital or states are passed, so the engine splits the country itself
# and there is no state-ownership geometry to maintain.
POR_start_client_civil_war = {
	FROM = {
		POR_score_client = yes

		set_variable = { POR_cw_size = POR_net@ROOT }
		multiply_variable = { POR_cw_size = 0.55 }
		if = {
			limit = { ROOT = { has_country_flag = POR_via_aberta } }
			add_to_variable = { POR_cw_size = 0.08 }
		}
		if = {
			limit = { ROOT = { has_country_flag = POR_via_discreta } }
			subtract_from_variable = { POR_cw_size = 0.10 }
		}

		# Backing the weak and the outlawed buys a bigger position for less money, and the
		# price is that the rebellion itself is weaker. The officers we bought are what
		# make it winnable.
		set_variable = { POR_cw_underdog = ROOT.POR_reward_mult }
		subtract_from_variable = { POR_cw_underdog = 1 }
		multiply_variable = { POR_cw_underdog = 0.25 }
		subtract_from_variable = { POR_cw_size = POR_cw_underdog }
		clamp_variable = { var = POR_cw_size min = 0.15 max = 0.60 }

		set_variable = { POR_cw_army = POR_cw_size }
		add_to_variable = { POR_cw_army = POR_officers@ROOT }
		clamp_variable = { var = POR_cw_army min = 0.15 max = 0.75 }

		set_country_flag = { flag = POR_client_cooldown@ROOT days = 1825 value = 1 }
		set_variable = { POR_net@ROOT = 0.2 }
		set_variable = { POR_officers@ROOT = 0 }
	}

	if = {
		limit = { has_government = democratic }
		FROM = {
			start_civil_war = {
				ideology = democratic
				size = POR_cw_size
				army_ratio = POR_cw_army
				navy_ratio = POR_cw_size
				air_ratio = POR_cw_size
				set_country_flag = POR_client_fresh
				set_country_flag = POR_shadow_client
			}
			hidden_effect = { add_popularity = { ideology = democratic popularity = -0.5 } }
			recalculate_party = yes
			update_majority_threshold = yes
		}
		set_temp_variable = { POR_client_slot = 1 }
	}
	else_if = {
		limit = { has_government = communism }
		FROM = {
			start_civil_war = {
				ideology = communism
				size = POR_cw_size
				army_ratio = POR_cw_army
				navy_ratio = POR_cw_size
				air_ratio = POR_cw_size
				set_country_flag = POR_client_fresh
				set_country_flag = POR_shadow_client
			}
			hidden_effect = { add_popularity = { ideology = communism popularity = -0.5 } }
			recalculate_party = yes
			update_majority_threshold = yes
		}
		set_temp_variable = { POR_client_slot = 5 }
	}
	else_if = {
		limit = { has_government = nationalist }
		FROM = {
			start_civil_war = {
				ideology = nationalist
				size = POR_cw_size
				army_ratio = POR_cw_army
				navy_ratio = POR_cw_size
				air_ratio = POR_cw_size
				set_country_flag = POR_client_fresh
				set_country_flag = POR_shadow_client
			}
			hidden_effect = { add_popularity = { ideology = nationalist popularity = -0.5 } }
			recalculate_party = yes
			update_majority_threshold = yes
		}
		set_temp_variable = { POR_client_slot = 23 }
	}
	else = {
		FROM = {
			start_civil_war = {
				ideology = neutrality
				size = POR_cw_size
				army_ratio = POR_cw_army
				navy_ratio = POR_cw_size
				air_ratio = POR_cw_size
				set_country_flag = POR_client_fresh
				set_country_flag = POR_shadow_client
			}
			hidden_effect = { add_popularity = { ideology = neutrality popularity = -0.5 } }
			recalculate_party = yes
			update_majority_threshold = yes
		}
		set_temp_variable = { POR_client_slot = 14 }
	}

	POR_install_client_politics = yes
}

# Everything the new country needs to be a country. This runs outside the start_civil_war
# effect list, because the order between that list and on_declare_war's own politics copy
# is undocumented; doing it here is correct either way. Shape follows the only working
# fresh rebel setup in the mod, events/Influence.txt:1214-1270.
POR_install_client_politics = {
	every_country = {
		limit = { has_country_flag = POR_client_fresh }
		clr_country_flag = POR_client_fresh

		start_politics_input = yes
		set_variable = { ruling_party = ROOT.POR_client_slot }
		set_temp_variable = { _client_slot = ROOT.POR_client_slot }
		set_variable = { party_pop_array^_client_slot = 0.7 }

		if = {
			limit = { ROOT = { has_government = democratic } }
			set_popularities = {
				democratic = 70
				communism = 10
				neutrality = 10
				fascism = 0
				nationalist = 10
			}
			set_politics = {
				ruling_party = democratic
				elections_allowed = no
			}
		}
		else_if = {
			limit = { ROOT = { has_government = communism } }
			set_popularities = {
				democratic = 10
				communism = 70
				neutrality = 10
				fascism = 0
				nationalist = 10
			}
			set_politics = {
				ruling_party = communism
				elections_allowed = no
			}
		}
		else_if = {
			limit = { ROOT = { has_government = nationalist } }
			set_popularities = {
				democratic = 10
				communism = 10
				neutrality = 10
				fascism = 0
				nationalist = 70
			}
			set_politics = {
				ruling_party = nationalist
				elections_allowed = no
			}
		}
		else = {
			set_popularities = {
				democratic = 10
				communism = 10
				neutrality = 70
				fascism = 0
				nationalist = 10
			}
			set_politics = {
				ruling_party = neutrality
				elections_allowed = no
			}
		}

		hidden_effect = {
			startup_politics = yes
			set_leader = yes
			recalculate_party = yes
			hold_elections = yes
		}

		init_influence = yes
		set_variable = { domestic_influence_amount = 70 }
		add_to_array = { influence_array = ROOT }
		add_to_array = { influence_array_val = 30 }
		sort_influence = yes

		add_opinion_modifier = { target = ROOT modifier = supported_us_in_the_coup }

		set_country_flag = POR_client_pending_puppet
		set_variable = { POR_client_overlord_id = ROOT.id }

		country_event = { id = portugal.30 hours = 6 }
	}
}

# Run on a client that has just won its war. A dynamic tag owns its territory without
# holding cores on any of it, which leaves it permanently in a compliance crisis.
POR_client_core_repair = {
	every_owned_state = {
		limit = { NOT = { is_core_of = PREV } }
		add_core_of = PREV
	}
	un_recog_setup_check = yes
}

# The money lines only appear while they are actually paying or costing something.
POR_update_money_ideas = {
	if = {
		limit = {
			check_variable = { POR_lusophone_subjects^num > 0 }
			NOT = { has_idea = POR_ultramar_income_money }
		}
		add_ideas = POR_ultramar_income_money
	}
	else_if = {
		limit = {
			check_variable = { POR_lusophone_subjects^num < 1 }
			has_idea = POR_ultramar_income_money
		}
		remove_ideas = POR_ultramar_income_money
	}

	if = {
		limit = {
			OR = {
				check_variable = { POR_active_operations > 0 }
				check_variable = { POR_lusophone_subjects^num > 0 }
			}
			NOT = { has_idea = POR_ultramar_expense_money }
		}
		add_ideas = POR_ultramar_expense_money
	}
	else_if = {
		limit = {
			check_variable = { POR_active_operations < 1 }
			check_variable = { POR_lusophone_subjects^num < 1 }
			has_idea = POR_ultramar_expense_money
		}
		remove_ideas = POR_ultramar_expense_money
	}

	if = {
		limit = {
			has_completed_focus = POR_o_instituto_camoes
			NOT = { has_idea = POR_lusofonia_money }
		}
		add_ideas = POR_lusofonia_money
	}

	if = {
		limit = {
			check_variable = { POR_intervention_count > 0 }
			NOT = { has_idea = POR_legiao_money }
		}
		add_ideas = POR_legiao_money
	}
	else_if = {
		limit = {
			check_variable = { POR_intervention_count < 1 }
			has_idea = POR_legiao_money
		}
		remove_ideas = POR_legiao_money
	}
}

# The ladder. What a country gives us long before it is ours: first a trading position,
# then the industry, then the officers. Reconciled weekly rather than bought with a
# decision, so the benefit always matches the network we actually hold there.
POR_apply_client_tier = {
	set_temp_variable = { POR_tier_wanted = 0 }
	if = { limit = { check_variable = { POR_net@ROOT > 0.699 } } set_temp_variable = { POR_tier_wanted = 3 } }
	else_if = { limit = { check_variable = { POR_net@ROOT > 0.399 } } set_temp_variable = { POR_tier_wanted = 2 } }
	else_if = { limit = { check_variable = { POR_net@ROOT > 0.199 } } set_temp_variable = { POR_tier_wanted = 1 } }

	if = {
		limit = {
			check_variable = { POR_tier_wanted = 0 }
			OR = {
				has_idea = POR_presenca_lusofona_1
				has_idea = POR_presenca_lusofona_2
				has_idea = POR_presenca_lusofona_3
			}
		}
		remove_ideas = { POR_presenca_lusofona_1 POR_presenca_lusofona_2 POR_presenca_lusofona_3 }
	}
	else_if = {
		limit = {
			check_variable = { POR_tier_wanted = 1 }
			NOT = { has_idea = POR_presenca_lusofona_1 }
		}
		remove_ideas = { POR_presenca_lusofona_2 POR_presenca_lusofona_3 }
		add_ideas = POR_presenca_lusofona_1
	}
	else_if = {
		limit = {
			check_variable = { POR_tier_wanted = 2 }
			NOT = { has_idea = POR_presenca_lusofona_2 }
		}
		remove_ideas = { POR_presenca_lusofona_1 POR_presenca_lusofona_3 }
		add_ideas = POR_presenca_lusofona_2
	}
	else_if = {
		limit = {
			check_variable = { POR_tier_wanted = 3 }
			NOT = { has_idea = POR_presenca_lusofona_3 }
		}
		remove_ideas = { POR_presenca_lusofona_1 POR_presenca_lusofona_2 }
		add_ideas = POR_presenca_lusofona_3
	}
}

# The price of distance. Every theatre Portuguese troops are standing in has to be fuelled
# and supplied from Lisbon, and a good share of what is landed is stolen, sold or written
# off long before it reaches the firing line. The first intervention already hurts; the
# second is what makes an expensive army unaffordable and a cheap one the obvious answer.
POR_update_intervention_load = {
	set_variable = { POR_intervention_count = nations_we_sent_volunteers_to^num }

	if = {
		limit = { check_variable = { POR_intervention_count < 1 } }
		clear_variable = POR_intervention_fuel
		clear_variable = POR_intervention_upkeep
		clear_variable = POR_intervention_supply
		if = {
			limit = { has_dynamic_modifier = { modifier = POR_intervencao_ultramarina } }
			remove_dynamic_modifier = { modifier = POR_intervencao_ultramarina }
		}
	}
	else = {
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = POR_intervencao_ultramarina } } }

# Ideas spreading through the public argument. A ratchet: it never falls, because an
# argument the country has already had cannot be un-had. POR_ideo_add is the input.
POR_raise_ideological_development = {
	if = {
		limit = { check_variable = { POR_ideo_add = 0 } }
		set_temp_variable = { POR_ideo_add = 2 }
	}
	# The raw variable change prints an empty "which grants:" block whenever the tier it
	# feeds is still worth nothing, so the whole thing is tooltipped by hand.
	custom_effect_tooltip = POR_ideological_development_up_tt
	hidden_effect = {
		add_to_variable = { POR_ideological_development = POR_ideo_add }
		clamp_variable = { var = POR_ideological_development min = 0 max = 100 }
		clear_variable = POR_ideo_add
		POR_update_ideology_tier = yes
	}
}

POR_update_ideology_tier = {
	if = {
		limit = {
			check_variable = { POR_ideological_development > 0 }
			NOT = { has_dynamic_modifier = { modifier = POR_ideologia_modifier } }
		}
		add_dynamic_modifier = { modifier = POR_ideologia_modifier }
	}

	if = { limit = { check_variable = { POR_ideological_development > 79 } } set_variable = { POR_ideology_tier = 4 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 59 } } set_variable = { POR_ideology_tier = 3 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 39 } } set_variable = { POR_ideology_tier = 2 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 19 } } set_variable = { POR_ideology_tier = 1 } }
	else = { set_variable = { POR_ideology_tier = 0 } }

	# The wider the argument, the easier the country is to move. Recomputed weekly into the same
	# field the ideology focuses write, so last week's contribution comes back out first.
	if = {
		limit = { NOT = { has_variable = POR_ideologia_tier_drift } }
		set_variable = { POR_ideologia_tier_drift = 0 }
	}
	subtract_from_variable = { POR_ideologia_drift_defence_factor = POR_ideologia_tier_drift }
	set_variable = { POR_ideologia_tier_drift = POR_ideology_tier }
	multiply_variable = { POR_ideologia_tier_drift = -0.03 }
	add_to_variable = { POR_ideologia_drift_defence_factor = POR_ideologia_tier_drift }
}

# A change of government is not a reset. Every time the country argues itself into handing
# power to the other side, the argument itself gets wider. on_government_change cannot
# carry this: PS, PSD and the liberals all sit in the democratic group
# (00_MD_politicsview_scripted_effects.txt:184-193), so a normal alternation never fires it.
POR_watch_party_flip = {
	set_party_index_to_ruling_party = yes

	if = {
		limit = { NOT = { has_variable = POR_last_ruling_party } }
		set_variable = { POR_last_ruling_party = party_index }
	}
	if = {
		limit = { NOT = { check_variable = { POR_last_ruling_party = party_index } } }
		set_temp_variable = { POR_ideo_add = 5 }
		# Crossing an outlook group is a bigger shock than an alternation inside one.
		if = {
			limit = {
				OR = {
					AND = {
						check_variable = { POR_last_ruling_party < 4 }
						check_variable = { party_index > 3 }
					}
					AND = {
						check_variable = { POR_last_ruling_party > 3 }
						check_variable = { party_index < 4 }
					}
				}
			}
			set_temp_variable = { POR_ideo_add = 8 }
		}
		POR_raise_ideological_development = yes
		add_to_variable = { POR_party_flips = 1 }
		news_event = { id = portugal_news.7 hours = 6 }
	}
	set_variable = { POR_last_ruling_party = party_index }
	clear_variable = party_index
}

# A standing pull toward whoever is in office. Fascism has no drift field on the ideology
# modifier and pulls the nationalist bloc instead.
POR_pull_towards_ruling_outlook = {
	if = {
		limit = { has_government = communism }
		add_to_variable = { POR_ideologia_communism_drift = 0.05 tooltip = communism_drift_tt }
	}
	else_if = {
		limit = { OR = { has_government = nationalist has_government = fascism } }
		add_to_variable = { POR_ideologia_nationalist_drift = 0.05 tooltip = nationalist_drift_tt }
	}
	else_if = {
		limit = { has_government = neutrality }
		add_to_variable = { POR_ideologia_neutrality_drift = 0.05 tooltip = neutrality_drift_tt }
	}
	else = {
		add_to_variable = { POR_ideologia_democratic_drift = 0.05 tooltip = democratic_drift_tt }
	}
}

# Weekly snapshot of where support sits. Focus availability has to be a trigger, and none
# of this is readable as one, so it is cached here in the same shape as the bloc totals.
POR_update_ideology_support = {
	set_party_index_to_ruling_party = yes
	set_temp_variable = { outlook_party_index = party_index }
	set_outlook_from_party_index = yes
	set_variable = { POR_ruling_outlook = outlook_id }
	set_outlook_popularity_from_id = yes
	set_variable = { POR_ruling_outlook_support = outlook_popularity }

	# The strongest outlook that is not in office. This is what would take power if the
	# argument ever stopped being an argument.
	set_outlook_popularity_array = yes
	set_temp_variable = { POR_rev_best = 0 }
	set_temp_variable = { POR_rev_best_id = -1 }
	for_loop_effect = {
		start = 0
		end = 5
		value = POR_o
		if = {
			limit = {
				NOT = { check_variable = { POR_o = POR_ruling_outlook } }
				check_variable = { outlook_popularity_by_id^POR_o > POR_rev_best }
			}
			set_temp_variable = { POR_rev_best = outlook_popularity_by_id^POR_o }
			set_temp_variable = { POR_rev_best_id = POR_o }
		}
	}
	set_variable = { POR_revolt_support = POR_rev_best }
	set_variable = { POR_revolt_outlook = POR_rev_best_id }

	# Its largest party, which is who ends up in the chair.
	set_temp_variable = { POR_slot_start = 0 }
	set_temp_variable = { POR_slot_end = 0 }
	if = {
		limit = { check_variable = { POR_revolt_outlook = 0 } }
		set_temp_variable = { POR_slot_start = 0 }
		set_temp_variable = { POR_slot_end = 4 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 1 } }
		set_temp_variable = { POR_slot_start = 4 }
		set_temp_variable = { POR_slot_end = 10 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 2 } }
		set_temp_variable = { POR_slot_start = 10 }
		set_temp_variable = { POR_slot_end = 12 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 3 } }
		set_temp_variable = { POR_slot_start = 12 }
		set_temp_variable = { POR_slot_end = 20 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 4 } }
		set_temp_variable = { POR_slot_start = 20 }
		set_temp_variable = { POR_slot_end = 24 }
	}

	set_temp_variable = { POR_rev_party_pop = 0 }
	set_temp_variable = { POR_rev_party_id = -1 }
	for_loop_effect = {
		start = POR_slot_start
		end = POR_slot_end
		value = POR_p
		if = {
			limit = { check_variable = { party_pop_array^POR_p > POR_rev_party_pop } }
			set_temp_variable = { POR_rev_party_pop = party_pop_array^POR_p }
			set_temp_variable = { POR_rev_party_id = POR_p }
		}
	}
	set_variable = { POR_revolt_party = POR_rev_party_id }

	# The largest party out of office, for the focuses that poach its support.
	set_temp_variable = { POR_opp_pop = 0 }
	set_temp_variable = { POR_opp_id = -1 }
	for_loop_effect = {
		start = 0
		end = 24
		value = POR_q
		if = {
			limit = {
				NOT = { check_variable = { POR_q = party_index } }
				check_variable = { party_pop_array^POR_q > POR_opp_pop }
			}
			set_temp_variable = { POR_opp_pop = party_pop_array^POR_q }
			set_temp_variable = { POR_opp_id = POR_q }
		}
	}
	set_variable = { POR_opposition_party = POR_opp_id }
	clear_variable = party_index
}

# The revolution: the leading outlook out of office takes the chair, and the one it
# displaced goes into the hills. The displaced side is read before the handover, because
# change_ruling_party_effect rewrites everything this needs.
POR_run_revolution = {

	set_temp_variable = { POR_rev_outlook_old = POR_ruling_outlook }
	set_temp_variable = { POR_rev_size = POR_ruling_outlook_support }
	multiply_temp_variable = { POR_rev_size = 0.8 }

	set_temp_variable = { change_leader_temp = 0 }
	set_temp_variable = { disable_elections = 0 }
	change_ruling_party_effect = yes

	# A fascism spawn hands the rebels to the jihadist chain in 00_on_actions.txt, so the
	# displaced far right revolts as nationalists.
	set_country_flag = no_leader_change
	if = {
		limit = { check_variable = { POR_rev_outlook_old = 0 } }
		start_civil_war = { ideology = democratic size = POR_rev_size }
	}
	else_if = {
		limit = { check_variable = { POR_rev_outlook_old = 1 } }
		start_civil_war = { ideology = communism size = POR_rev_size }
	}
	else_if = {
		limit = { check_variable = { POR_rev_outlook_old = 3 } }
		start_civil_war = { ideology = neutrality size = POR_rev_size }
	}
	else = {
		start_civil_war = { ideology = nationalist size = POR_rev_size }
	}
	clr_country_flag = no_leader_change

	add_stability = -0.20
	add_war_support = 0.10
	set_temp_variable = { int_investment_change = -20 }
	modify_international_investment_effect = yes
	set_temp_variable = { debt_change = 20 }
	modify_debt_effect = yes
	set_variable = { protest_strength = 0 }
	set_temp_variable = { POR_ideo_add = 15 }
	POR_raise_ideological_development = yes
	news_event = { id = portugal_news.8 hours = 6 }

# A landing to take the Portuguese nationals off the beach, which happens to leave a
# Portuguese garrison on it. The state remembers who owned it, so the bill can arrive when
# the fighting stops. Called from POR_evacuar_os_nossos with FROM as the host.
POR_take_evacuation_state = {
	FROM = {
		random_owned_controlled_state = {
			limit = {
				is_capital = no
				is_coastal = yes
				NOT = { has_state_flag = POR_evacuation_state }
			}
			set_state_flag = POR_evacuation_state
			set_variable = { POR_evacuation_owner = PREV.id }
			ROOT = { transfer_state = PREV }
		}
		add_to_array = { ROOT.POR_evacuation_array = THIS.id }
		add_opinion_modifier = { target = ROOT modifier = border_tensions }
	}
	set_temp_variable = { POR_heat_add = 10 }
	POR_add_exposure = yes
}

# Weekly: the fighting has stopped somewhere we landed, so who keeps the beach comes due.
# One at a time, and the claimant arrives on the event as FROM rather than in a variable
# that a second landing could overwrite.
POR_check_evacuations = {
	if = {
		limit = { check_variable = { POR_evacuation_array^num > 0 } }
		random_scope_in_array = {
			array = POR_evacuation_array
			limit = {
				OR = {
					NOT = { exists = yes }
					AND = { has_war = no has_civil_war = no }
				}
			}
			ROOT = { remove_from_array = { POR_evacuation_array = PREV.id } }
			if = {
				limit = {
					exists = yes
					NOT = { has_country_flag = POR_shadow_client }
					NOT = { is_subject_of = ROOT }
				}
				ROOT = { country_event = { id = portugal.42 days = 7 } }
			}
			else = {
				ROOT = { country_event = { id = portugal.41 days = 7 } }
			}
		}
	}
}

# Hand the beach back. FROM is the country asking for it.
POR_return_evacuation_state = {
	every_owned_state = {
		limit = {
			has_state_flag = POR_evacuation_state
			check_variable = { POR_evacuation_owner = FROM.id }
		}
		clr_state_flag = POR_evacuation_state
		clear_variable = POR_evacuation_owner
		FROM = { transfer_state = PREV }
	}
	FROM = {
		remove_opinion_modifier = { target = ROOT modifier = border_tensions }
		add_opinion_modifier = { target = ROOT modifier = eased_border_tensions }
	}
	set_temp_variable = { POR_heat_add = -8 }
	POR_add_exposure = yes
}

# Keep it. Everyone concerned understands what happened, including the people who signed the
# evacuation request.
POR_keep_evacuation_state = {

		clear_variable = POR_evacuation_owner
		add_core_of = ROOT

	}
	add_war_support = 0.05
	set_temp_variable = { POR_heat_add = 15 }
	POR_add_exposure = yes
}

# Backing a side in somebody else's civil war.
#
# The commitment is recorded as a flag on each side rather than as a tag id held by Portugal,
# because a rebel is a dynamic tag that may not survive the war, and because on_civil_war_end
# hands us the winner as ROOT and annexes the loser
# (common/on_actions/05_lar_on_actions.txt:975). A flag travels with the country scope whatever
# happens to the tag, so "did our side win" is answered by looking at the winner alone.
#
# The price of a commitment is paid when it is made and the reward when it is won, and the
# reward lands on the winner - influence, a network and standing there - rather than as a bonus
# at home. The multiplier is fixed at commitment time and stored on the backed side, so backing
# an underdog is priced the day it is done and a second commitment elsewhere cannot overwrite it.

# All three run in the scope of the country concerned, with ROOT as Portugal. Every caller is a
# focus completion or an event option, where ROOT is Portugal already, so nothing has to be
# threaded through FROM.

# The side Portugal is backing.
POR_mark_backed = {
	set_country_flag = POR_backed_by_portugal
	clr_country_flag = POR_opposed_by_portugal
	add_opinion_modifier = { target = ROOT modifier = supported_our_claim }
}

# The side Portugal is backing against.
POR_mark_opposed = {
	set_country_flag = POR_opposed_by_portugal
	clr_country_flag = POR_backed_by_portugal
	add_opinion_modifier = { target = ROOT modifier = supported_opposition }
}

# The commitment itself, run on the side being backed. The underdog multiplier is scored and
# stored here, on this country, so a later commitment somewhere else cannot overwrite it.
# Takes POR_commit_heat on Portugal.
POR_commit_here = {
	POR_score_client = yes
	set_variable = { POR_backing_mult@ROOT = ROOT.POR_reward_mult }
	POR_mark_backed = yes

	ROOT = {
		add_to_variable = { POR_backing_count = 1 }
		if = {
			limit = { check_variable = { POR_commit_heat = 0 } }
			set_temp_variable = { POR_commit_heat = 10 }
		}
		set_temp_variable = { POR_heat_add = POR_commit_heat }
		POR_add_exposure = yes
		clear_variable = POR_commit_heat
		custom_effect_tooltip = POR_commit_to_a_side_tt
	}
}

# Called from on_civil_war_end in the WINNER's scope. Portugal is a third party here, so
# anything of ours runs inside an explicit POR scope.
POR_resolve_backing = {
	if = {
		limit = {
			has_country_flag = POR_backed_by_portugal
			country_exists = POR
		}
		clr_country_flag = POR_backed_by_portugal
		if = {
			limit = { NOT = { has_variable = POR_backing_mult@POR } }
			set_variable = { POR_backing_mult@POR = 1 }
		}
		set_temp_variable = { POR_pay = 12 }
		multiply_temp_variable = { POR_pay = POR_backing_mult@POR }
		round_temp_variable = POR_pay

		add_to_variable = { POR_net@POR = 0.25 }
		clamp_variable = { var = POR_net@POR min = 0 max = 1 }
		add_opinion_modifier = { target = POR modifier = ideological_alliance }
		reverse_add_opinion_modifier = { target = POR modifier = ideological_alliance }

		POR = {
			set_temp_variable = { percent_change = PREV.POR_pay }
			set_temp_variable = { tag_index = THIS.id }
			set_temp_variable = { influence_target = PREV.id }
			change_influence_percentage = yes
			add_to_variable = { POR_int_trade_opinion_factor = 0.05 tooltip = trade_opinion_factor_tt }
			subtract_from_variable = { POR_backing_count = 1 }
			clamp_variable = { var = POR_backing_count min = 0 max = 20 }
			add_political_power = 50
			if = {
				limit = { NOT = { is_in_array = { POR_network_target_array = PREV.id } } }
				add_to_array = { POR_network_target_array = PREV.id }
			}
			country_event = { id = portugal.70 hours = 6 }
		}
		clear_variable = POR_backing_mult@POR
	}
	else_if = {
		limit = {
			has_country_flag = POR_opposed_by_portugal
			country_exists = POR
		}
		clr_country_flag = POR_opposed_by_portugal
		set_variable = { POR_net@POR = 0 }
		add_opinion_modifier = { target = POR modifier = supported_opposition }
		POR = {
			set_temp_variable = { percent_change = -10 }
			set_temp_variable = { tag_index = THIS.id }
			set_temp_variable = { influence_target = PREV.id }
			change_influence_percentage = yes
			subtract_from_variable = { POR_backing_count = 1 }
			clamp_variable = { var = POR_backing_count min = 0 max = 20 }
			set_temp_variable = { POR_heat_add = 12 }
			POR_add_exposure = yes
			country_event = { id = portugal.71 hours = 6 }
		}
		clear_variable = POR_backing_mult@POR
	}
}
