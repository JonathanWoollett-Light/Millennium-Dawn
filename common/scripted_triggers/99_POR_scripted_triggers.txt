# AI path ownership. Internal to ai_will_do, so no tooltips and no loc keys.

POR_ai_alt_path = {
	OR = {
		has_global_flag = POR_LUSOPHONE_FOCUS_PATH
		has_global_flag = POR_IMPERIAL_FOCUS_PATH
	}
}

POR_ai_not_lusophone_path = {
	NOT = { has_global_flag = POR_LUSOPHONE_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = POR_HISTORICAL_FOCUS_PATH
		has_global_flag = POR_IMPERIAL_FOCUS_PATH
		POR_ai_not_shadow_path = yes
	}
}

POR_ai_not_imperial_path = {
	NOT = { has_global_flag = POR_IMPERIAL_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = POR_HISTORICAL_FOCUS_PATH
		has_global_flag = POR_LUSOPHONE_FOCUS_PATH
		POR_ai_not_shadow_path = yes
	}
}

# Both alt paths run the Lusophone spine; only the imperial one takes it to a war.
POR_ai_shadow_path = {
	POR_ai_alt_path = yes
}

POR_ai_not_shadow_path = {
	NOT = { POR_ai_alt_path = yes }
}

# Parliamentary blocs. The totals are recomputed weekly from the frozen election result by
# POR_update_blocs; majority_threshold is half the valid vote and is written by the same
# core effects (00_MD_politicsview_scripted_effects.txt:964).
POR_bloc_left_has_majority = {
	check_variable = { POR_bloc_left > majority_threshold }
}

POR_bloc_right_has_majority = {
	check_variable = { POR_bloc_right > majority_threshold }
}

POR_bloc_centre_has_majority = {
	check_variable = { POR_bloc_centre > majority_threshold }
}

# How far the public argument has moved. Radical policy is gated on these.
POR_ideology_tier_1 = {
	check_variable = { POR_ideological_development > 19 }
}

POR_ideology_tier_2 = {
	check_variable = { POR_ideological_development > 39 }
}

POR_ideology_tier_3 = {
	check_variable = { POR_ideological_development > 59 }
}

POR_ideology_tier_4 = {
	check_variable = { POR_ideological_development > 79 }
}

# Whether the government has a base worth organising, and whether anyone else does.
POR_ruling_party_has_support = {
	check_variable = { party_pop_array^ruling_party > 0.15 }
}

POR_has_an_opposition = {
	check_variable = { POR_opposition_party > -1 }
}

# The strongest outlook out of office, cached weekly by POR_update_ideology_support.
POR_revolt_has_support = {
	check_variable = { POR_revolt_party > -1 }
	check_variable = { POR_revolt_support > 0.34 }
}

# Which family holds office. The alignment focuses read these; the socialists sit in the
# Atlantic camp rather than the left one, which is the whole quarrel of the period.
POR_the_left_governs = {
	OR = {
		check_variable = { ruling_party = 4 }
		check_variable = { ruling_party = 5 }
		check_variable = { ruling_party = 18 }
		check_variable = { ruling_party = 19 }
	}
}

POR_the_atlanticists_govern = {
	OR = {
		check_variable = { ruling_party = 1 }
		check_variable = { ruling_party = 2 }
		check_variable = { ruling_party = 3 }
	}
}
