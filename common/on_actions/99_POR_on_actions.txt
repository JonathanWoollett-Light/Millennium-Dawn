on_actions = {
	# ROOT is the winner, FROM is the side that was annexed. Puppeting waits until here:
	# an overlord relationship declared during the war would block volunteers and leave
	# the client unable to finish it (99_SOV_on_actions.txt:143).
	on_civil_war_end = {
		effect = {
			# ROOT is the winner here, so a side Portugal backed is identified by the
			# flag it still carries.
			POR_resolve_backing = yes

			if = {
				limit = { has_country_flag = POR_client_pending_puppet }
				clr_country_flag = POR_client_pending_puppet
				POR_client_core_repair = yes
				news_event = { id = portugal_news.2 hours = 6 }
				var:POR_client_overlord_id = {
					set_autonomy = {
						target = PREV
						autonomy_state = autonomy_associated_state
						end_wars = no
						end_civil_wars = yes
					}
					country_event = { id = portugal.32 hours = 8 }
				}
				clear_variable = POR_client_overlord_id
			}

			# Our side lost. The file on it closes, and the government that survived knows
			# exactly who paid for the attempt.
			if = {
				limit = {
					FROM = { has_country_flag = POR_shadow_client }
					NOT = { has_country_flag = POR_shadow_client }
				}
				FROM = {
					if = {
						limit = { check_variable = { POR_client_overlord_id > 0 } }
						var:POR_client_overlord_id = { country_event = { id = portugal.34 hours = 8 } }
					}
				}
				add_opinion_modifier = { target = var:FROM.POR_client_overlord_id modifier = supported_opposition }
			}
		}
	}

	on_weekly_POR = {
		effect = {
			POR_update_blocs = yes
			POR_watch_party_flip = yes
			POR_update_ideology_tier = yes
			POR_update_ideology_support = yes

			POR_prune_shadow_arrays = yes
			POR_update_intervention_load = yes

			# Walk every country we are working on up or down the ladder to match the
			# network we actually hold there.
			for_each_scope_loop = {
				array = POR_network_target_array
				POR_apply_client_tier = yes
			}

			POR_update_overseas_doctrine = yes
			POR_update_money_ideas = yes

			# Exposure only cools while nothing of ours is fighting overseas, and it cools
			# slower the more heat we have left lying around in client states. The per-target
			# heat cools on its own, so a country we stop working eventually goes quiet.
			set_temp_variable = { POR_heat_total = 0 }
			for_each_scope_loop = {
				array = POR_network_target_array
				if = {
					limit = { check_variable = { POR_heat@ROOT > 0 } }
					ROOT = { add_to_temp_variable = { POR_heat_total = PREV.POR_heat@ROOT } }
					subtract_from_variable = { POR_heat@ROOT = 0.1 }
					clamp_variable = { var = POR_heat@ROOT min = 0 max = 100 }
				}
			}
			if = {
				limit = {
					check_variable = { POR_active_operations = 0 }
					check_variable = { POR_intervention_count = 0 }
					check_variable = { POR_shadow_exposure > 0 }
				}
				set_temp_variable = { POR_cooling = 0.5 }
				if = {
					limit = { check_variable = { POR_heat_total > 19 } }
					set_temp_variable = { POR_cooling = 0.1 }
				}
				else_if = {
					limit = { check_variable = { POR_heat_total > 9 } }
					set_temp_variable = { POR_cooling = 0.25 }
				}
				subtract_from_variable = { POR_shadow_exposure = POR_cooling }
			}

			POR_check_evacuations = yes
			POR_update_exposure_tier = yes
		}
	}
}
