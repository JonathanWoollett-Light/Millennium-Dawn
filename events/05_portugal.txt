# The client government is shown a file about what its neighbour is said to be planning.
# FROM is Portugal.
country_event = {
	id = portugal.43
	is_triggered_only = yes
	hidden = yes

	# ROOT is the client government, FROM is Portugal.
	immediate = {
		random_neighbor_country = {
			limit = {
				exists = yes
				is_subject = no
				has_war = no
				NOT = { tag = FROM }
				NOT = { tag = ROOT }
				NOT = { is_in_faction_with = ROOT }
				any_owned_state = { is_controlled_by = PREV }
			}
			add_opinion_modifier = { target = ROOT modifier = border_tensions }
			# The claim has to exist before the war goal does: take_claimed_state draws its
			# states from claims, and take_state_focus without a generator draws none at all.
			random_owned_state = {
				limit = { is_controlled_by = PREV }
				add_claim_by = ROOT
			}
			ROOT = { create_wargoal = { type = take_claimed_state target = PREV } }
		}
	}
}
