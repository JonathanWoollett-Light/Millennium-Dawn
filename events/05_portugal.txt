# The target notices. Forty-five days between the order and the shooting is long enough to
# act on, if anyone in the capital is still reading the reports.
country_event = {
	id = portugal.20
	title = portugal.20.t
	desc = portugal.20.d
	picture = GFX_SAO_coup

	is_triggered_only = yes

	# Purge the officer corps
	option = {
		name = portugal.20.a
		log = "[GetDateText]: [This.GetName]: portugal.20.a executed"

		add_stability = -0.10
		subtract_from_variable = { POR_net@FROM = 0.20 }
		clamp_variable = { var = POR_net@FROM min = 0 max = 1 }
		set_variable = { POR_officers@FROM = 0 }

		set_temp_variable = { party_index = ruling_party }
		set_temp_variable = { party_popularity_increase = 0.05 }
		change_relative_party_popularity = yes

		ai_chance = {
			base = 40
			modifier = {
				factor = 3
				check_variable = { POR_net@FROM > 0.8 }
			}
			modifier = {
				factor = 0.25
				has_stability < 0.4
			}
		}
	}

	# There is nothing to it
	option = {
		name = portugal.20.b

		ai_chance = {
			base = 60
			modifier = {
				factor = 0.5
				check_variable = { POR_net@FROM > 0.8 }
			}
		}
	}
}

	id = portugal.32
	title = portugal.32.t
	desc = portugal.32.d
	picture = GFX_SAO_Election

	is_triggered_only = yes

	option = {
		name = portugal.32.a
		log = "[GetDateText]: [This.GetName]: portugal.32.a executed"

		add_to_array = { POR_lusophone_subjects = FROM.id }
		remove_from_array = { POR_shadow_clients = FROM.id }
		set_variable = { POR_active_operations = POR_shadow_clients^num }

		# The autonomy terms replace the informal arrangement. Leaving both in place would
		# charge the client twice for the same trade.
		FROM = {
			remove_ideas = { POR_presenca_lusofona_1 POR_presenca_lusofona_2 POR_presenca_lusofona_3 }
		}

		set_temp_variable = { POR_heat_add = 6 }
		FROM = { POR_score_client = yes }
		POR_add_exposure = yes
		POR_update_exposure_tier = yes
	}
}

	id = portugal.60
	title = portugal.60.t
	desc = portugal.60.d
	picture = GFX_SAO_Election

	is_triggered_only = yes

	# Accede to the federation
	option = {
		name = portugal.60.a
		log = "[GetDateText]: [This.GetName]: portugal.60.a executed"

		set_country_flag = POR_federacao_willing
		set_country_flag = POR_federacao_decided
		FROM = { country_event = { id = portugal.61 hours = 6 } }

		ai_chance = {
			base = 20
			modifier = {
				factor = 2
				has_opinion = { target = FROM value > 50 }
			}
			modifier = {
				factor = 3
				check_variable = { POR_federacao_pressure > 0 }
			}
			modifier = {
				factor = 0.5
				has_war = yes
			}
		}
	}

	# Remain sovereign
	option = {
		name = portugal.60.b

		set_country_flag = POR_federacao_decided

		ai_chance = {
			base = 30
			modifier = {
				factor = 0.5
				check_variable = { POR_federacao_pressure > 1 }
			}
		}
	}
}
